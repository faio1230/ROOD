cmake_minimum_required(VERSION 3.21)

project(rood VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(ROOD_BUILD_GUI "Build the Qt Widgets shell when Qt 6 is available" ON)
option(ROOD_REQUIRE_QT "Fail configuration when Qt 6 Widgets is unavailable" OFF)
option(ROOD_BUILD_PORTAUDIO_PROBE "Build the PortAudio device probe when PortAudio is available" ON)
option(ROOD_BUILD_MEDIA_DEPENDENCY_PROBE "Link and inspect the pinned media development libraries" OFF)
option(ROOD_BUILD_SRT_INGEST "Build the independent SRT receive and FFmpeg decode engine" OFF)
option(ROOD_BUILD_AUDIO_OUTPUT "Build PortAudio output for decoded SRT audio" OFF)
option(ROOD_BUILD_SPOUT_OUTPUT "Build scheduled Spout2 video output" OFF)
option(ROOD_BUILD_OMT_OUTPUT "Build timestamped OMT video and routed audio output" OFF)

add_library(rood_engine STATIC
    src/engine/ChannelRouter.cpp
    src/engine/AudioTimeline.cpp
)
target_include_directories(rood_engine PUBLIC include)
target_compile_features(rood_engine PUBLIC cxx_std_17)

if(MSVC)
    target_compile_options(rood_engine PRIVATE /W4)
else()
    target_compile_options(rood_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(ROOD_BUILD_SRT_INGEST)
    find_path(ROOD_FFMPEG_INCLUDE_DIR libavformat/avformat.h REQUIRED)
    find_library(ROOD_AVFORMAT_LIBRARY NAMES avformat REQUIRED)
    find_library(ROOD_AVCODEC_LIBRARY NAMES avcodec REQUIRED)
    find_library(ROOD_AVUTIL_LIBRARY NAMES avutil REQUIRED)
    find_path(ROOD_LIBSRT_INCLUDE_DIR srt/srt.h REQUIRED)
    find_library(ROOD_LIBSRT_LIBRARY NAMES srt REQUIRED)

    add_library(rood_media STATIC src/media/MediaReceiver.cpp)
    target_include_directories(rood_media PUBLIC include "${ROOD_FFMPEG_INCLUDE_DIR}" "${ROOD_LIBSRT_INCLUDE_DIR}")
    target_link_libraries(rood_media PUBLIC
        "${ROOD_AVFORMAT_LIBRARY}" "${ROOD_AVCODEC_LIBRARY}"
        "${ROOD_AVUTIL_LIBRARY}" "${ROOD_LIBSRT_LIBRARY}")
    if(WIN32)
        target_link_libraries(rood_media PUBLIC ws2_32)
    endif()
    if(MSVC)
        target_compile_options(rood_media PRIVATE /W4 /utf-8)
    else()
        target_compile_options(rood_media PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_executable(rood_ingest apps/ingest/main.cpp)
    target_link_libraries(rood_ingest PRIVATE rood_media)
    if(MSVC)
        target_compile_options(rood_ingest PRIVATE /W4 /utf-8)
    endif()
endif()

if(ROOD_BUILD_AUDIO_OUTPUT)
    if(NOT TARGET rood_media)
        message(FATAL_ERROR "ROOD_BUILD_AUDIO_OUTPUT requires ROOD_BUILD_SRT_INGEST")
    endif()
    find_path(PORTAUDIO_INCLUDE_DIR portaudio.h REQUIRED)
    find_library(PORTAUDIO_LIBRARY NAMES portaudio portaudio_x64 REQUIRED)
    find_library(ROOD_SWRESAMPLE_LIBRARY NAMES swresample REQUIRED)
    if(WIN32)
        find_path(PORTAUDIO_WASAPI_INCLUDE_DIR pa_win_wasapi.h HINTS "${PORTAUDIO_INCLUDE_DIR}" REQUIRED)
    endif()
    add_library(rood_audio STATIC src/audio/PortAudioOutput.cpp)
    target_include_directories(rood_audio PUBLIC include PRIVATE
        "${PORTAUDIO_INCLUDE_DIR}" "${PORTAUDIO_WASAPI_INCLUDE_DIR}")
    target_link_libraries(rood_audio PUBLIC rood_engine rood_media
        "${PORTAUDIO_LIBRARY}" "${ROOD_SWRESAMPLE_LIBRARY}")
    if(MSVC)
        target_compile_options(rood_audio PRIVATE /W4 /utf-8)
    else()
        target_compile_options(rood_audio PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    target_link_libraries(rood_ingest PRIVATE rood_audio)
    target_compile_definitions(rood_ingest PRIVATE ROOD_HAS_AUDIO_OUTPUT=1)
endif()

if(ROOD_BUILD_SPOUT_OUTPUT)
    if(NOT WIN32 OR NOT TARGET rood_media)
        message(FATAL_ERROR "ROOD_BUILD_SPOUT_OUTPUT requires Windows and ROOD_BUILD_SRT_INGEST")
    endif()
    find_library(ROOD_SWSCALE_LIBRARY NAMES swscale REQUIRED)
    find_path(ROOD_SPOUT_INCLUDE_DIR SpoutLibrary/SpoutLibrary.h REQUIRED)
    find_library(ROOD_SPOUT_LIBRARY NAMES SpoutLibrary REQUIRED)
    add_library(rood_video STATIC src/video/SpoutVideoOutput.cpp)
    target_include_directories(rood_video PUBLIC include PRIVATE "${ROOD_SPOUT_INCLUDE_DIR}")
    target_link_libraries(rood_video PUBLIC rood_media
        "${ROOD_SWSCALE_LIBRARY}" "${ROOD_SPOUT_LIBRARY}")
    if(MSVC)
        target_compile_options(rood_video PRIVATE /W4 /utf-8)
    endif()
    target_link_libraries(rood_ingest PRIVATE rood_video)
    target_compile_definitions(rood_ingest PRIVATE ROOD_HAS_SPOUT_OUTPUT=1)

    add_executable(rood_spout_probe apps/spout_probe/main.cpp)
    target_include_directories(rood_spout_probe PRIVATE "${ROOD_SPOUT_INCLUDE_DIR}")
    target_link_libraries(rood_spout_probe PRIVATE "${ROOD_SPOUT_LIBRARY}")
    if(MSVC)
        target_compile_options(rood_spout_probe PRIVATE /W4 /utf-8)
    endif()
endif()

if(ROOD_BUILD_OMT_OUTPUT)
    if(NOT TARGET rood_media)
        message(FATAL_ERROR "ROOD_BUILD_OMT_OUTPUT requires ROOD_BUILD_SRT_INGEST")
    endif()
    find_path(ROOD_OMT_INCLUDE_DIR libomt.h HINTS "${ROOD_OMT_DIR}" REQUIRED)
    find_library(ROOD_OMT_LIBRARY NAMES libomt HINTS "${ROOD_OMT_DIR}" REQUIRED)
    find_library(ROOD_SWRESAMPLE_LIBRARY NAMES swresample REQUIRED)
    find_library(ROOD_SWSCALE_LIBRARY NAMES swscale REQUIRED)
    add_library(rood_omt STATIC src/omt/OmtOutput.cpp)
    target_include_directories(rood_omt PUBLIC include PRIVATE "${ROOD_OMT_INCLUDE_DIR}")
    target_link_libraries(rood_omt PUBLIC rood_engine rood_media
        "${ROOD_OMT_LIBRARY}" "${ROOD_SWRESAMPLE_LIBRARY}" "${ROOD_SWSCALE_LIBRARY}")
    if(MSVC)
        target_compile_options(rood_omt PRIVATE /W4 /utf-8)
    endif()
    target_link_libraries(rood_ingest PRIVATE rood_omt)
    target_compile_definitions(rood_ingest PRIVATE ROOD_HAS_OMT_OUTPUT=1)

    add_executable(rood_omt_probe apps/omt_probe/main.cpp)
    target_include_directories(rood_omt_probe PRIVATE "${ROOD_OMT_INCLUDE_DIR}")
    target_link_libraries(rood_omt_probe PRIVATE "${ROOD_OMT_LIBRARY}")
    if(MSVC)
        target_compile_options(rood_omt_probe PRIVATE /W4 /utf-8)
    endif()
endif()

add_executable(rood_diag apps/diag/main.cpp)
target_link_libraries(rood_diag PRIVATE rood_engine)

if(ROOD_BUILD_GUI)
    find_package(Qt6 QUIET COMPONENTS Widgets)
    if(Qt6Widgets_FOUND)
        add_executable(rood_gui WIN32 apps/gui/main.cpp)
        target_link_libraries(rood_gui PRIVATE rood_engine Qt6::Widgets)
        if(TARGET rood_media AND TARGET rood_audio AND TARGET rood_video AND TARGET rood_omt)
            target_sources(rood_gui PRIVATE apps/gui/MediaWindow.cpp)
            target_link_libraries(rood_gui PRIVATE rood_media rood_audio rood_video rood_omt)
            target_compile_definitions(rood_gui PRIVATE ROOD_GUI_HAS_MEDIA=1)
        endif()
        if(MSVC)
            target_compile_options(rood_gui PRIVATE /utf-8)
        endif()
        message(STATUS "Qt Widgets GUI: enabled (${Qt6_VERSION})")
    else()
        if(ROOD_REQUIRE_QT)
            message(FATAL_ERROR "Qt 6 Widgets is required but was not found. Set Qt6_DIR or CMAKE_PREFIX_PATH.")
        endif()
        message(STATUS "Qt 6 Widgets not found; GUI target skipped. Set Qt6_DIR or CMAKE_PREFIX_PATH after installing Qt.")
    endif()
endif()

if(ROOD_BUILD_PORTAUDIO_PROBE)
    find_path(PORTAUDIO_INCLUDE_DIR portaudio.h)
    find_library(PORTAUDIO_LIBRARY NAMES portaudio portaudio_x64)
    if(WIN32)
        find_path(PORTAUDIO_WASAPI_INCLUDE_DIR pa_win_wasapi.h HINTS "${PORTAUDIO_INCLUDE_DIR}")
    endif()
    if(PORTAUDIO_INCLUDE_DIR AND PORTAUDIO_LIBRARY AND
       (NOT WIN32 OR PORTAUDIO_WASAPI_INCLUDE_DIR))
        add_executable(rood_pa_probe apps/pa_probe/main.cpp)
        target_include_directories(rood_pa_probe PRIVATE "${PORTAUDIO_INCLUDE_DIR}")
        if(WIN32)
            target_include_directories(rood_pa_probe PRIVATE "${PORTAUDIO_WASAPI_INCLUDE_DIR}")
        endif()
        target_link_libraries(rood_pa_probe PRIVATE "${PORTAUDIO_LIBRARY}")
        message(STATUS "PortAudio device probe: enabled (${PORTAUDIO_LIBRARY})")
    else()
        message(STATUS "PortAudio development files not found; probe target skipped.")
    endif()
endif()

if(ROOD_BUILD_MEDIA_DEPENDENCY_PROBE)
    find_path(ROOD_FFMPEG_INCLUDE_DIR libavcodec/avcodec.h REQUIRED)
    find_library(ROOD_AVCODEC_LIBRARY NAMES avcodec REQUIRED)
    find_library(ROOD_AVUTIL_LIBRARY NAMES avutil REQUIRED)
    find_path(ROOD_LIBSRT_INCLUDE_DIR srt/srt.h REQUIRED)
    find_library(ROOD_LIBSRT_LIBRARY NAMES srt REQUIRED)
    find_path(ROOD_SPOUT_INCLUDE_DIR SpoutLibrary/SpoutLibrary.h REQUIRED)
    find_library(ROOD_SPOUT_LIBRARY NAMES SpoutLibrary REQUIRED)
    find_path(ROOD_OMT_INCLUDE_DIR libomt.h HINTS "${ROOD_OMT_DIR}" REQUIRED)
    find_library(ROOD_OMT_LIBRARY NAMES libomt HINTS "${ROOD_OMT_DIR}" REQUIRED)

    add_executable(rood_deps_probe apps/deps_probe/main.cpp)
    target_include_directories(rood_deps_probe PRIVATE
        "${ROOD_FFMPEG_INCLUDE_DIR}"
        "${ROOD_LIBSRT_INCLUDE_DIR}"
        "${ROOD_SPOUT_INCLUDE_DIR}"
        "${ROOD_OMT_INCLUDE_DIR}")
    target_link_libraries(rood_deps_probe PRIVATE
        "${ROOD_AVCODEC_LIBRARY}"
        "${ROOD_AVUTIL_LIBRARY}"
        "${ROOD_LIBSRT_LIBRARY}"
        "${ROOD_SPOUT_LIBRARY}"
        "${ROOD_OMT_LIBRARY}")
    if(MSVC)
        target_compile_options(rood_deps_probe PRIVATE /utf-8)
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(channel_router_test tests/ChannelRouterTest.cpp)
    target_link_libraries(channel_router_test PRIVATE rood_engine)
    add_test(NAME channel_router COMMAND channel_router_test)
    add_executable(audio_timeline_test tests/AudioTimelineTest.cpp)
    target_link_libraries(audio_timeline_test PRIVATE rood_engine)
    add_test(NAME audio_timeline COMMAND audio_timeline_test)
endif()

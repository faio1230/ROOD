cmake_minimum_required(VERSION 3.21)

project(srt_receiver VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(SRT_RX_BUILD_GUI "Build the Qt Widgets shell when Qt 6 is available" ON)
option(SRT_RX_REQUIRE_QT "Fail configuration when Qt 6 Widgets is unavailable" OFF)
option(SRT_RX_BUILD_PORTAUDIO_PROBE "Build the PortAudio device probe when PortAudio is available" ON)

add_library(srt_rx_engine STATIC
    src/engine/ChannelRouter.cpp
)
target_include_directories(srt_rx_engine PUBLIC include)
target_compile_features(srt_rx_engine PUBLIC cxx_std_17)

if(MSVC)
    target_compile_options(srt_rx_engine PRIVATE /W4)
else()
    target_compile_options(srt_rx_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(srt_rx_diag apps/diag/main.cpp)
target_link_libraries(srt_rx_diag PRIVATE srt_rx_engine)

if(SRT_RX_BUILD_GUI)
    find_package(Qt6 QUIET COMPONENTS Widgets)
    if(Qt6Widgets_FOUND)
        add_executable(srt_rx_gui WIN32 apps/gui/main.cpp)
        target_link_libraries(srt_rx_gui PRIVATE srt_rx_engine Qt6::Widgets)
        if(MSVC)
            target_compile_options(srt_rx_gui PRIVATE /utf-8)
        endif()
        message(STATUS "Qt Widgets GUI: enabled (${Qt6_VERSION})")
    else()
        if(SRT_RX_REQUIRE_QT)
            message(FATAL_ERROR "Qt 6 Widgets is required but was not found. Set Qt6_DIR or CMAKE_PREFIX_PATH.")
        endif()
        message(STATUS "Qt 6 Widgets not found; GUI target skipped. Set Qt6_DIR or CMAKE_PREFIX_PATH after installing Qt.")
    endif()
endif()

if(SRT_RX_BUILD_PORTAUDIO_PROBE)
    find_path(PORTAUDIO_INCLUDE_DIR portaudio.h)
    find_library(PORTAUDIO_LIBRARY NAMES portaudio portaudio_x64)
    if(WIN32)
        find_path(PORTAUDIO_WASAPI_INCLUDE_DIR pa_win_wasapi.h HINTS "${PORTAUDIO_INCLUDE_DIR}")
    endif()
    if(PORTAUDIO_INCLUDE_DIR AND PORTAUDIO_LIBRARY AND
       (NOT WIN32 OR PORTAUDIO_WASAPI_INCLUDE_DIR))
        add_executable(srt_rx_pa_probe apps/pa_probe/main.cpp)
        target_include_directories(srt_rx_pa_probe PRIVATE "${PORTAUDIO_INCLUDE_DIR}")
        if(WIN32)
            target_include_directories(srt_rx_pa_probe PRIVATE "${PORTAUDIO_WASAPI_INCLUDE_DIR}")
        endif()
        target_link_libraries(srt_rx_pa_probe PRIVATE "${PORTAUDIO_LIBRARY}")
        message(STATUS "PortAudio device probe: enabled (${PORTAUDIO_LIBRARY})")
    else()
        message(STATUS "PortAudio development files not found; probe target skipped.")
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(channel_router_test tests/ChannelRouterTest.cpp)
    target_link_libraries(channel_router_test PRIVATE srt_rx_engine)
    add_test(NAME channel_router COMMAND channel_router_test)
endif()
